/* Original, dependency-free georeferenced terrain preview. Heights stay at 1:1 scale.
   Canvas compatibility renderer: actual DEM mesh, camera-relative lighting and height-field shadows.
   Uses request-driven rendering; full mesh at rest, coarser mesh only while interacting. */
export class TerrainRenderer {
 constructor(canvas,grid,options={}){
  if(!grid||grid.width<2||grid.height<2||grid.elevations.length!==grid.width*grid.height)throw new Error('Invalid elevation grid');
  this.canvas=canvas;this.ctx=canvas.getContext('2d',{alpha:false});if(!this.ctx)throw new Error('Canvas unavailable');
  this.grid=grid;this.options=options;this.yaw=-.48;this.pitch=.73;this.zoom=1.7;this.offset=[0,0];this.sun=0;this.destroyed=false;this.dragging=false;this.frame=0;this.pointers=new Map();this.route=[];this.quality='full';
  const b=grid.bounds;this.widthM=(b[2]-b[0])*111320*Math.cos((b[1]+b[3])/2*Math.PI/180);this.heightM=(b[3]-b[1])*111320;this.base=(grid.minElevation+grid.maxElevation)/2;
  this.vertices=grid.elevations.map((h,i)=>({x:(i%grid.width/(grid.width-1)-.5)*this.widthM,y:(.5-Math.floor(i/grid.width)/(grid.height-1))*this.heightM,z:h-this.base,h}));
  for(let i=0;i<this.vertices.length;i++){const x=i%grid.width,y=Math.floor(i/grid.width),left=this.vertices[y*grid.width+Math.max(0,x-1)],right=this.vertices[y*grid.width+Math.min(grid.width-1,x+1)],north=this.vertices[Math.max(0,y-1)*grid.width+x],south=this.vertices[Math.min(grid.height-1,y+1)*grid.width+x],n=[-(right.z-left.z)/(right.x-left.x),-(north.z-south.z)/(north.y-south.y),1],length=Math.hypot(...n);this.vertices[i].normal=n.map(a=>a/length);}
  this.meshes={1:this.buildMesh(1),2:this.buildMesh(2),4:this.buildMesh(4)};this.shadows=new Map();this.events=[];
  this.bind('pointerdown',e=>{this.canvas.setPointerCapture(e.pointerId);this.pointers.set(e.pointerId,[e.clientX,e.clientY]);this.dragging=true;this.request();});
  this.bind('pointermove',e=>{const old=this.pointers.get(e.pointerId);if(!old)return;this.pointers.set(e.pointerId,[e.clientX,e.clientY]);if(this.pointers.size===1){this.yaw+=(e.clientX-old[0])*.004;this.pitch=Math.max(.3,Math.min(1.2,this.pitch+(e.clientY-old[1])*.0025));}else{const p=[...this.pointers.values()],a=p[0],b=p[1],distance=Math.hypot(a[0]-b[0],a[1]-b[1]);if(this.pinch)this.zoom=Math.max(.8,Math.min(5,this.zoom*distance/this.pinch));this.pinch=distance;}this.request();});
  const up=e=>{this.pointers.delete(e.pointerId);this.pinch=0;if(!this.pointers.size){this.dragging=false;this.request();}};this.bind('pointerup',up);this.bind('pointercancel',up);
  this.bind('wheel',e=>{e.preventDefault();this.zoom=Math.max(.8,Math.min(5,this.zoom*Math.exp(-e.deltaY*.001)));this.dragging=true;clearTimeout(this.wheelTimer);this.wheelTimer=setTimeout(()=>{this.dragging=false;this.request();},120);this.request();},{passive:false});
  this.bind('keydown',e=>{if(e.key==='ArrowLeft')this.yaw-=.1;else if(e.key==='ArrowRight')this.yaw+=.1;else if(e.key==='ArrowUp')this.pitch=Math.min(1.2,this.pitch+.05);else if(e.key==='ArrowDown')this.pitch=Math.max(.3,this.pitch-.05);else if(e.key==='+'||e.key==='=')this.zoom=Math.min(5,this.zoom*1.15);else if(e.key==='-')this.zoom=Math.max(.8,this.zoom/1.15);else if(e.key==='Home'){this.yaw=-.48;this.pitch=.73;this.zoom=1.7;}else return;e.preventDefault();this.request();});
  if(typeof ResizeObserver!=='undefined'){this.resize=new ResizeObserver(()=>this.request());this.resize.observe(canvas);}else{this.onResize=()=>this.request();window.addEventListener('resize',this.onResize);}this.request();
 }
 bind(name,fn,options){this.canvas.addEventListener(name,fn,options);this.events.push([name,fn,options]);}
 heightAt(x,y){const W=this.grid.width,H=this.grid.height,gx=(x/this.widthM+.5)*(W-1),gy=(.5-y/this.heightM)*(H-1);if(gx<0||gy<0||gx>W-1||gy>H-1)return null;const ix=Math.min(W-2,Math.floor(gx)),iy=Math.min(H-2,Math.floor(gy)),a=gx-ix,b=gy-iy,v=this.grid.elevations;return(v[iy*W+ix]*(1-a)+v[iy*W+ix+1]*a)*(1-b)+(v[(iy+1)*W+ix]*(1-a)+v[(iy+1)*W+ix+1]*a)*b;}
 buildMesh(step){const W=this.grid.width,H=this.grid.height,list=[],v=this.vertices;for(let y=0;y<H-1;y+=step)for(let x=0;x<W-1;x+=step){const xx=Math.min(x+step,W-1),yy=Math.min(y+step,H-1),a=y*W+x,b=y*W+xx,c=yy*W+x,d=yy*W+xx;for(const ids of [[a,c,b],[b,c,d]]){const p=v[ids[0]],q=v[ids[1]],r=v[ids[2]],u=[q.x-p.x,q.y-p.y,q.z-p.z],w=[r.x-p.x,r.y-p.y,r.z-p.z];let normal=[u[1]*w[2]-u[2]*w[1],u[2]*w[0]-u[0]*w[2],u[0]*w[1]-u[1]*w[0]];if(normal[2]<0)normal=normal.map(n=>-n);const length=Math.hypot(...normal);normal=[0,1,2].map(axis=>(p.normal[axis]+q.normal[axis]+r.normal[axis])/3);const smoothLength=Math.hypot(...normal);normal=normal.map(n=>n/smoothLength);list.push({ids,n:normal,x:(p.x+q.x+r.x)/3,y:(p.y+q.y+r.y)/3,h:(p.h+q.h+r.h)/3});}}return list;}
 setGeometry(data){const lines=[];const visit=g=>{if(!g)return;if(g.type==='FeatureCollection')g.features.forEach(visit);else if(g.type==='Feature')visit(g.geometry);else if(g.type==='GeometryCollection')g.geometries.forEach(visit);else if(g.type==='LineString')lines.push(g.coordinates);else if(g.type==='MultiLineString')lines.push(...g.coordinates);};visit(data);const b=this.grid.bounds;this.route=[];for(const line of lines){let prev=null;for(let i=0;i<line.length;i++){const ll=line[i],x=((ll[0]-b[0])/(b[2]-b[0])-.5)*this.widthM,y=(.5-(b[3]-ll[1])/(b[3]-b[1]))*this.heightM,h=this.heightAt(x,y);if(h===null){prev=null;continue;}const p={x,y,z:h-this.base+8};if(prev&&Math.hypot(p.x-prev.x,p.y-prev.y)>6){this.route.push([prev,p]);prev=p;}else if(!prev)prev=p;}}this.request();}
 color(height,light,fog){const stops=[[0,[70,98,79]],[1000,[103,127,91]],[1600,[144,159,111]],[2050,[178,181,141]],[2600,[188,183,163]],[3300,[227,225,211]]];let a=stops[0],b=stops[stops.length-1];for(let i=0;i<stops.length-1;i++)if(height>=stops[i][0]&&height<stops[i+1][0]){a=stops[i];b=stops[i+1];break;}const t=Math.max(0,Math.min(1,(height-a[0])/(b[0]-a[0]||1))),mist=[216,228,220];const c=a[1].map((v,i)=>{const base=v+(b[1][i]-v)*t;return Math.round(Math.min(255,base*light)*(1-fog)+mist[i]*fog);});return`rgb(${c.join(',')})`;}
 shadowAt(triangle,light){const step=420,slope=light[2]/Math.hypot(light[0],light[1]);for(let d=step;d<5500;d+=step){const h=this.heightAt(triangle.x+light[0]*d,triangle.y+light[1]*d);if(h===null)break;if(h>triangle.h+slope*d+12)return .74;}return 1;}
 project(p){const cr=Math.cos(this.yaw),sr=Math.sin(this.yaw),cp=Math.cos(this.pitch),sp=Math.sin(this.pitch),x=p.x*cr-p.y*sr,y=p.x*sr+p.y*cr,depth=y*cp-p.z*sp,dist=Math.max(this.widthM,this.heightM)*2.2,per=dist/(dist+depth);return{x:this.cx+x*this.scale*per,y:this.cy+(-y*sp-p.z*cp)*this.scale*per,d:depth};}
 request(){if(this.destroyed||this.frame)return;this.frame=requestAnimationFrame(()=>{this.frame=0;if(!this.destroyed)this.render();});}
 render(){const started=performance.now(),c=this.canvas,ctx=this.ctx,rect=c.getBoundingClientRect();if(rect.width<1||rect.height<1)return;const W=rect.width,H=rect.height,mobile=W<650,dpr=Math.min(devicePixelRatio||1,this.dragging?1.5:2);if(c.width!==Math.round(W*dpr)||c.height!==Math.round(H*dpr)){c.width=Math.round(W*dpr);c.height=Math.round(H*dpr);}ctx.setTransform(dpr,0,0,dpr,0,0);this.cx=W*.52;this.cy=H*(this.options.compactViewport&&mobile ? .25 : .57);const usableH=this.options.compactViewport&&mobile?H*.5:H;this.scale=Math.min(W/this.widthM,usableH*1.65/this.heightM)*this.zoom;
  const sky=ctx.createLinearGradient(0,0,0,H);sky.addColorStop(0,'#c3d9d5');sky.addColorStop(.4,'#e1e7d8');sky.addColorStop(1,'#e7e9d9');ctx.fillStyle=sky;ctx.fillRect(0,0,W,H);
  const step=this.dragging?2:1,mesh=this.meshes[step],projected=this.vertices.map(p=>this.project(p)),sunAngle=-this.yaw+2.4,sunAltitude=.7+this.pitch*.12,light=[Math.cos(sunAngle)*Math.cos(sunAltitude),Math.sin(sunAngle)*Math.cos(sunAltitude),Math.sin(sunAltitude)];const lightBucket=Math.round(sunAngle*8)+':'+Math.round(sunAltitude*8)+':'+step;
  let shadows=this.shadows.get(lightBucket);if(!shadows&&!this.dragging){shadows=mesh.map(t=>this.shadowAt(t,light));if(this.shadows.size>8)this.shadows.delete(this.shadows.keys().next().value);this.shadows.set(lightBucket,shadows);}
  const items=mesh.map((t,i)=>({t,i,d:(projected[t.ids[0]].d+projected[t.ids[1]].d+projected[t.ids[2]].d)/3}));for(const segment of this.route){const a=this.project(segment[0]),b=this.project(segment[1]);items.push({line:[a,b],d:(a.d+b.d)/2-2});}items.sort((a,b)=>b.d-a.d);
  for(const item of items){if(item.line){const[a,b]=item.line;ctx.beginPath();ctx.moveTo(a.x,a.y);ctx.lineTo(b.x,b.y);ctx.lineCap='round';ctx.strokeStyle='#bf622f';ctx.lineWidth=2.8;ctx.shadowColor='#fff9d9';ctx.shadowBlur=2;ctx.stroke();ctx.shadowBlur=0;continue;}const t=item.t,a=projected[t.ids[0]],b=projected[t.ids[1]],d=projected[t.ids[2]];if(Math.max(a.x,b.x,d.x)<-2||Math.min(a.x,b.x,d.x)>W+2||Math.max(a.y,b.y,d.y)<-2||Math.min(a.y,b.y,d.y)>H+2)continue;const diffuse=Math.max(0,t.n[0]*light[0]+t.n[1]*light[1]+t.n[2]*light[2]);const illumination=(.43+diffuse*.72)*(shadows?.[item.i]??1),fog=Math.max(0,Math.min(.38,(item.d/Math.max(this.widthM,this.heightM)+.15)*.44));ctx.fillStyle=this.color(t.h,illumination,fog);ctx.strokeStyle=ctx.fillStyle;ctx.lineWidth=.45;ctx.beginPath();ctx.moveTo(a.x,a.y);ctx.lineTo(b.x,b.y);ctx.lineTo(d.x,d.y);ctx.closePath();ctx.fill();ctx.stroke();}
  // A restrained orientation dial, tied to camera bearing; no geographic labels are invented.
  ctx.save();ctx.translate(38,usableH-42);ctx.rotate(-this.yaw);ctx.beginPath();ctx.moveTo(0,-12);ctx.lineTo(5,7);ctx.lineTo(0,3);ctx.lineTo(-5,7);ctx.closePath();ctx.fillStyle='#213c30';ctx.fill();ctx.restore();ctx.font='600 12px Gotham Pro, Manrope, system-ui';ctx.fillStyle='#405c48';ctx.fillText('С',34,usableH-62);
  c.dataset.renderMs=(performance.now()-started).toFixed(1);c.dataset.triangles=String(mesh.length);c.dataset.quality=this.dragging?'motion':'full';c.dataset.renderer='dem-mesh-canvas';this.options.onRender?.({ms:performance.now()-started,triangles:mesh.length,quality:this.dragging?'motion':'full'});
 }
 rotate(delta){this.yaw+=delta;this.request();}
 zoomBy(factor){this.zoom=Math.max(.8,Math.min(5,this.zoom*factor));this.request();}
 reset(){this.yaw=-.48;this.pitch=.73;this.zoom=1.7;this.request();}
 destroy(){this.destroyed=true;cancelAnimationFrame(this.frame);clearTimeout(this.wheelTimer);this.resize?.disconnect();if(this.onResize)window.removeEventListener('resize',this.onResize);this.events.forEach(([name,fn,options])=>this.canvas.removeEventListener(name,fn,options));}
}
